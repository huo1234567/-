class UndoStack:
    def __init__(self):
        self.stack = []
        self.top = -1

    def push(self, item):
        self.top += 1
        self.stack.append(item)

    def pop(self):

        del self.stack[self.top]
        self.top -= 1

class HistoryQueue:
    def __init__(self, capacity=10):
        self.queue = []
        self.capacity = capacity

    def enqueue(self, item):
        if len(self.queue) >= self.capacity:
            self.dequeue()
        self.queue.append(item)

    def dequeue(self):
        if not self.queue:
            return None
        return self.queue.pop(0)

    def remove_last(self):
        if self.queue:
            self.queue.pop()

    def recent(self):
        return list(self.queue)

def top_k_with_score(word: str, k: int, doc_tokens: dict[int, list[str]]):
    word = word.lower()
    scores = []
    for doc_id, tokens in doc_tokens.items():
        cnt = tokens.count(word)
        if cnt > 0:
            scores.append((-cnt, doc_id, cnt))
    scores.sort()
    top_list = [(sid, cnt) for (_, sid, cnt) in scores[:k]]
    return top_list

def top_k_heap_with_score(word: str, k: int, doc_tokens: dict[int, list[str]]):
    word = word.lower()
    heap = []
    for doc_id, tokens in doc_tokens.items():
        cnt = tokens.count(word)
        if cnt > 0:
            if len(heap) < k:
                heapq.heappush(heap, (cnt, doc_id))
            else:
                if cnt > heap[0][0]:
                    heapq.heappop(heap)
                    heapq.heappush(heap, (cnt, doc_id))
    heap_sorted = sorted(heap, key=lambda x: (-x[0], x[1]))
    return [(sid, cnt) for cnt, sid in heap_sorted]

# ==================== 功能5：拼写纠错（动态规划 编辑距离） ====================
def edit_distance(s1: str, s2: str) -> int:
    """
    动态规划实现最小编辑距离（二维DP表）
    允许操作：插入、删除、替换
    """
    m, n = len(s1), len(s2)
    # 创建二维 DP 表
    dp = [[0] * (n + 1) for _ in range(m + 1)]

    # 初始化边界
    for i in range(m + 1):
        dp[i][0] = i
    for j in range(n + 1):
        dp[0][j] = j

    # 填充 DP 表
    for i in range(1, m + 1):
        for j in range(1, n + 1):
            if s1[i-1] == s2[j-1]:
                dp[i][j] = dp[i-1][j-1]
            else:
                dp[i][j] = 1 + min(
                    dp[i-1][j],    # 删除
                    dp[i][j-1],    # 插入
                    dp[i-1][j-1]   # 替换
                )
    return dp[m][n]

def spell_correct(query: str, vocab: set) -> str | None:
    """
    找到索引中编辑距离最小的英文单词
    仅对英文纠错，中文直接返回 None
    """
    query = query.lower()
    # 只对纯英文单词纠错
    if not query.isalpha() or any('\u4e00' <= c <= '\u9fff' for c in query):
        return None

    min_dist = float('inf')
    best_word = None

    for word in vocab:
        # 只对比英文单词
        if not word.isalpha():
            continue
        dist = edit_distance(query, word)
        if dist < min_dist:
            min_dist = dist
            best_word = word
        elif dist == min_dist:
            # 距离相同，选字典序小的
            if word < best_word:
                best_word = word
    return best_word

        print("========== 系统功能菜单 ==========")
        print("1. 普通查询：直接输入查询词")
        print("2. Top-K相关性查询：topk:词:k:A  /  topk:词:k:B")
        print("3. 撤销上一次查询：输入 undo")
        print("4. 查看最近历史记录：输入 history")
        print("5. 显示全部文档切词结果：输入 show")
        print("6. 退出系统：输入 exit")
        print("================================\n")

        while True:
            user_input = input("请输入指令/查询词：").strip()

            if user_input.lower() == "exit":
                print("系统已退出")
                break

            elif user_input.lower() == "undo":
                canceled_word = undo_stack.pop()
                if canceled_word is None:
                    print("→ 无操作可撤销\n")
                    continue
                history_queue.remove_last()
                canceled_res = search_word(canceled_word, inverted_index)
                print(f"→ 已撤销上一次查询：[{canceled_word}] → {canceled_res}\n")
                continue

            elif user_input.lower() == "history":
                print(f"最近10次查询记录（早→晚）：{history_queue.recent()}\n")
                continue

            elif user_input.lower() == "show":
                print("\n========== 全部文档切词结果 ==========")
                for doc_id, tokens in doc_dict.items():
                    print(f"文档{doc_id} 词项列表：{tokens}")
                    print(f"文档{doc_id} 词项总数：{len(tokens)}\n")
                continue

            elif user_input.startswith("topk:"):
                parts = user_input.split(":")
                if len(parts) != 4:
                    print("→ 格式错误！示例：topk:算:3:A\n")
                    continue
                try:
                    word = parts[1]
                    k = int(parts[2])
                    mode = parts[3].upper()
                    if mode == "A":
                        res = top_k_with_score(word, k, doc_dict)
                        print("→ 方案A Top-K 结果：")
                    elif mode == "B":
                        res = top_k_heap_with_score(word, k, doc_dict)
                        print("→ 方案B Top-K 结果：")
                    else:
                        print("→ 仅支持 A / B 两种方案\n")
                        continue
                    for doc_id, score in res:
                        print(f"   文档 {doc_id} (相关度: {score})")
                    print()
                    undo_stack.push(user_input)
                    history_queue.enqueue(user_input)
                except:
                    print("→ 输入格式错误！\n")
                continue

            if user_input:
                res = search_word(user_input, inverted_index)
                if res:
                    print(f"→ 搜索结果：[{user_input}] → {res}\n")
                    undo_stack.push(user_input)
                    history_queue.enqueue(user_input)
                else:
                    suggest = spell_correct(user_input, vocab)
                    if suggest:
                        print(f"→ 未找到该词，是否要查询：{suggest} ？\n")
                    else:
                        print(f"→ 未找到 [{user_input}]，无匹配内容\n")
    except FileNotFoundError:
        print("错误：未找到 testdata 文件夹，请检查目录！")
