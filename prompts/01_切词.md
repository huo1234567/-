import os
def tokenize_text(text: str) -> list[str]:
    """
    严格按照规则实现文本切词，不使用任何第三方库
    :param text: 原始文本字符串
    :return: 切词后的词项列表
    """
    tokens = []
    current_buffer = []  # 缓存连续的字母/数字
    text = text.lower()  # 统一转为小写
    for char in text:
        # 判断字符类型
        is_chinese = '\u4e00' <= char <= '\u9fff'  # 中文汉字
        is_letter = char.isalpha()                  # 英文字母
        is_digit = char.isdigit()                   # 数字
        if is_chinese:
            # 汉字：先清空缓存，单独作为词项
            if current_buffer:
                tokens.append(''.join(current_buffer))
                current_buffer.clear()
            tokens.append(char)
        elif is_letter or is_digit:
            # 字母/数字：加入缓存，连续组合
            current_buffer.append(char)
        else:
            # 分隔符：清空缓存，忽略该字符
            if current_buffer:
                tokens.append(''.join(current_buffer))
                current_buffer.clear()
    # 处理文本末尾剩余的缓存内容
    if current_buffer:
        tokens.append(''.join(current_buffer))
    return tokens
def load_and_tokenize(folder_path: str) -> dict[int, list[str]]:
    """
    批量读取文件夹下所有txt文件并完成切词（满足所有验收标准）
    :param folder_path: 文件夹路径（如 testdata/）
    :return: {文档编号(int): 切词列表(list[str])}，编号从0开始按文件名升序
    """
    # 获取所有txt文件并按文件名升序排序
    txt_files = [f for f in os.listdir(folder_path) if f.endswith('.txt')]
    txt_files.sort()  # 严格升序排列
    doc_tokens = {}
    for idx, filename in enumerate(txt_files):
        file_path = os.path.join(folder_path, filename)
        try:
            # 读取文件（兼容utf-8编码）
            with open(file_path, 'r', encoding='utf-8') as f:
                content = f.read()

            doc_tokens[idx] = tokens
        except Exception as e:
            print(f"读取文件 {filename} 失败：{str(e)}")
            doc_tokens[idx] = []  # 异常时返回空列表
    return doc_tokens
